🔹 Ansible — Create Docker Containers for Multiple Applications


---
- name: Setup docker container for nginx
  hosts: all
  become: yes

  tasks:
    - name: Pull nginx image
      docker_image: 
        name: nginx
        source: pull
    
    - name: Run Nginx Container
      docker_container: 
        name: nginx
        image: nginx
        state: started
        published_ports:
          - "8080:80"
