apiVersion: apps/v1
kind: StatefulSet
metadata:
  name: pathnex-db

spec:
  serviceName: "pathnex-db"

  # Creates:
  # pathnex-db-0
  # pathnex-db-1
  # pathnex-db-2
  replicas: 3

  selector:
    matchLabels:
      app: pathnex-db

  template:
    metadata:
      labels:
        app: pathnex-db

    spec:

      # Storage permissions for mounted volumes
      securityContext:
        fsGroup: 999

      containers:

        - name: db

          image: mysql:8.4

          ports:
            - containerPort: 3306

          resources:

            requests:
              # 250 millicores = 0.25 CPU
              cpu: "250m"
              memory: "512Mi"

            limits:
              cpu: "500m"
              memory: "1Gi"

          env:

            # Creates database during first initialization
            - name: MYSQL_DATABASE
              value: appdb

            # Password should come from Kubernetes Secret
            - name: MYSQL_ROOT_PASSWORD
              valueFrom:
                secretKeyRef:
                  name: mysql-secret
                  key: password


          volumeMounts:

            # MySQL data directory
            - name: mysql-storage
              mountPath: /var/lib/mysql


          # Checks if MySQL process is alive
          livenessProbe:

            tcpSocket:
              port: 3306

            initialDelaySeconds: 30
            periodSeconds: 10


          # Checks if MySQL is ready to accept connections
          readinessProbe:

            tcpSocket:
              port: 3306

            initialDelaySeconds: 10
            periodSeconds: 5


  # Creates one PVC per pod:
  #
  # mysql-storage-pathnex-db-0
  # mysql-storage-pathnex-db-1
  # mysql-storage-pathnex-db-2
  #
  volumeClaimTemplates:

    - metadata:
        name: mysql-storage

      spec:

        storageClassName: standard

        accessModes:
          - ReadWriteOnce

        resources:

          requests:
            storage: 20Gi

🔹 Jenkinsfile — Build and Deploy to Kubernetes
pipeline {
    agent any
    environment {
        IMAGE_NAME = 'pathnex-web-app'
    }
    stages {
        stage('Build Docker Image') {
            steps {
                script {
                    docker.build("${env.IMAGE_NAME}")
                }
            }
        }
        stage('Push Docker Image') {
            steps {
                script {
                    docker.withRegistry('https://docker.io', 'docker-credentials') {
                        docker.image("${env.IMAGE_NAME}").push('latest')
                    }
                }
            }
        }
        stage('Deploy to Kubernetes') {
            steps {
                script {
                    sh 'kubectl apply -f deployment.yaml'
                }
            }
        }
    }
}

on:
  push:
    branches:
      - main

jobs:
  build:
    runs-on: ubuntu-latest

    outputs: 
      image: pathnex-web-app
    
    steps:
      - name: Checkout code
        uses: actions/checkout@v4

      - name: Build Docker Image
        run: |
          docker build -t pathnex-web-app .

    
    push:
      needs: build
      runs-on: ubuntu-latest

      steps:
        - name: Checkout Code
          runs-on: ubuntu-latest

        - name: Login to container Registry
          run: |
            echo "${{ secrets.REGISTRY_PASSWORD }}" | docker login \
              -u "${{ secrets.REGISTRY_USERNAME }}" \
              --password-stdin

        - name: Build Docker Image
          run: |
            docker build -t pathnex-web-app .

        - name: Tag Image
          run: |
            docker tag pathnex-web-app ${{ secrets.REGISTRY_USERNAME }}/pathnex-web-app:latest

        - name: Push Image
          run: |
            docker push ${{ secrets.REGISTRY_USERNAME }}/pathnex-web-app:latest

    deploy:
      needs: push
      runs-on: ubuntu-latest

      steps:
        - name: Checkout Code
          uses: actions/checkout@v4

        - name: Configure Kubernetes
          run: |
            mkdir -p $HOME/.kube
            echo "${{ secrets.KUBE_CONFIG }}" | base64 -d > $HOME/.kube/config

        - name: Deploy to Kubernetes
          run: |
            kubectl apply -f kubernetes/deployment.yaml
