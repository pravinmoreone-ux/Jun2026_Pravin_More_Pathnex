🔹 Ansible — Implementing SSH Key-based Authentication


---
- name: Implementation SSH key based athentication
  hosts: all
  become: yes
  tasks:
    - name: Generate SSH key pair 
      command: ssh-keygen -t rsa -b 2048 -f /home/user/.ssh/id_rsa -N ""
    - name: Add ssh public key to authorized_key
      authorized_key:
        user: user
        state: present
        key: "{{ lookup('file', '/home/user/.ssh/id_rsa.pub') }}"

>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>>


| Option                 | Meaning                                |
| ---------------------- | -------------------------------------- |
| ssh-keygen             | Generate SSH keys                      |
| -t rsa                 | Use RSA algorithm                      |
| -b 2048                | Key size = 2048 bits                   |
| -f                     | Save key at this location              |
| /home/user/.ssh/id_rsa | Private key filename                   |
| -N ""                  | Empty passphrase (no password for key) |

🔹 Kubernetes — Network Policies for Security


apiVersion: networking.k8s.io/v1
kind: NetworkPolicy
metadata:
  name: pravin-web-policy
spec:
  podSelector:
    matchLabels:
      app: pravin-web-app
  policyTypes:
    - Ingress
    - Egress
  ingress:
    - from:
        - podSelector:
            matchLabels:
              app: pravin-api
  egress:
    - to:
        - podSelector:
            matchLabels:
              app: pravin-database

🔹 Jenkinsfile — Secure the Pipeline with Secrets
pipeline {
    agent any
    environment {
        AWS_ACCESS_KEY_ID = credentials('aws-access-key')
        AWS_SECRET_ACCESS_KEY = credentials('aws-secret-key')
    }
    stages {
        stage('Build') {
            steps {
                echo 'Building Docker image...'
                docker.build('pathnex-web-app')
            }
        }
        stage('Deploy') {
            steps {
                echo 'Deploying to AWS EC2...'
                sh 'aws ec2 run-instances --image-id ami-0abcd1234 --count 1 --instance-type t2.micro'
            }
        }
    }
}
