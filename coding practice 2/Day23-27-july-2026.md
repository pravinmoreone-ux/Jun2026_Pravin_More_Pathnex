Terraform serverless api gateway with lamdba integration

resource "aws_api_gateway_rest_api" "pravin_api" {
    name                = "pravin-api"
    description         = "Pravin API"
}

resource "aws_lambda_function" "pravin_lambda" {
    function_name = "pravin-lambda"
    runtime       = "python3.8"
    role          = aws_iam_role.lambda_exec_role.arn
    handler       = "lambda_function.lambda_handler"
    filename      = "lambda.zip"
}


resource "aws_api_gateway_integration" "lambda_integration" {
  rest_api_id = aws_api_gateway_rest_api.pravin_api.id
  resource_id = aws_api_gateway_resource.pravin_resource.id
  http_method = aws_api_gateway_method.pravin_method.http_method
  integration_http_method = "POST"
  type = "AWS_PROXY"
  uri  = aws_lambda_function.pravin_lambda.invoke_arn
}

🔹 Jenkinsfile — Deploy Serverless Applications with Lambda


pipeline {
    agent any

    stages {
        stage('Build') {
            steps {
                echo 'Building lambda funcion'
                sh 'zip -r lambda.zip lambda\'

            }
        }

        stage('Deploy to AWS Lambda') {
            steps {
                script {
                    sh 'aws lambda update-function-code --function-name pravin-lambda --zip-file --zip-file fileb://lambda.zip'
                }
            }
        }

    }
}

🔹 Docker
# Multi-Stage Docker Build


stage 1: 

FROM maven:3.9-eclipse-temurin-17 AS builder
WORKDIR /opt/pathnex/java-build
COPY . /opt/pathnex/java-build/
RUN mvn clean package

# Stage 2 — Production Stage
FROM tomcat:9
WORKDIR /usr/local/tomcat/webapps
COPY --from=builder /opt/pathnex/java-build/target/*.war /usr/local/tomcat/webapps/
EXPOSE 8080
CMD ["catalina.sh", "run"]

/opt/pathnex/java-build
