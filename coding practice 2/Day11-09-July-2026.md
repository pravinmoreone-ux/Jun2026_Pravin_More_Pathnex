🔹 Ansible — Install Helm on Server

---
- name: Install Helm on server
  hosts: all
  become: yes

  tasks:
    - name: Download Helm
      get_url: 
        url: https://get.helm.sh/helm-v3.7.0-linux-amd64.tar.gz
        dest: /tmp/helm.tar.gz

    - name: Extract Helm
      unarchive:
        src: /tmp/helm.tar.gz
        dest: /usr/local/bin/
        remote_src: yes

    - name: Set Helm binary permission
      file: 
        path: /usr/local/bin/helm
        mode: '0755'

    - name: Display Helm version

🔹 Jenkinsfile — Multi-Stage Pipeline



pipeline {
    agent any
    environment {
        IMAGE_NAME = 'pravin-web-app'
    }

    stages {
        stage('Checkout'){
            steps {
                git 'https://github.com/pravin/sample-repo.git'
            }
        }

        stage('Build Docker image') {
            script {
                docker.build("${env.IMAGE_NAME}")
            }            
        }

        stage('Push Docker Image'){
            steps{
                script{
                    docker.withRegistry('https://docker.io', 'docker-credentials') {
                        docker.image("${env.IMAGE_NAME}").push("latest")
                    }
                }
            }
        }

        stage('Deploy to kubernetes'){
           steps {
              script {
                sh 'kubectl apply -f deployment.yaml'
              }
           }     
        }
    }
}

Separate Nginx Ingress Controller Installation

The ingress controller should normally be installed once per Kubernetes cluster, not on every application deployment.


helm repo add ingress-nginx https://kubernetes.github.io/ingress-nginx

helm repo update

helm upgrade --install ingress-nginx \
  ingress-nginx/ingress-nginx \
  --namespace ingress-nginx \
  --create-namespace

  Then your application Helm chart only creates an Ingress resource:

apiVersion: networking.k8s.io/v1
kind: Ingress

metadata:
  name: pathnex-web-app

spec:
  ingressClassName: nginx

  rules:
    - host: pathnex.example.com
      http:
        paths:
          - path: /
            pathType: Prefix
            backend:
              service:
                name: pathnex-web-app
