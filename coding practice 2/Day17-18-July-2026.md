## 🔹 Ansible — Auto-healing Web Servers with Systemd



---
- name: Configure Auto - Healing for Web Server
  hosts: all
  become: yes
  tasks:
    - name: Ensure nginx server is running
      service: 
        name: nginx
        state: started
        enabled: yes
    
    - name: Setup auto-restart for nginx 
      lineinfile: 
        path: /etc/systemd/system/nginx.service.d/override.conf
        line: "Restart=always"
        create: yes
    
    - name: Configure Nginx restart policy
      blockinfile:
        path: /etc/systemd/system/nginx.service.d/override.conf
        create: yes
        block: |
            [Service]
            Restart=always
            RestartSec=5

    - name: reload systemd
      command: systemctl daemon-reload

🔹 Kubernetes — Self-Healing with Liveness and Readiness Probes

apiVersion: apps/v1
kind: Deployment
metadata:
  name: pravin-web-app
spec:
  replicas: 3
  selector:
    matchLabels:
      app: pravin-web-app
  template:
    metadata:
      labels:
        app: pravin-web-app
    spec:
      containers:
        - name: nginx
          image: nginx
          ports:
            - containerPort: 80
          livenessProbe:
            httpGet:
              path: /healthz
              port: 80
            initialDelaySeconds: 5
            periodSeconds: 10
          readinessProbe:
            httpGet:
              path: /healthz
              port: 80
            initialSeconds: 5
            periodSeconds: 10

🔹 Jenkinsfile — Rolling Deployments with Kubernetes


pipeline {
    agent any
    environment {
        IMAGE_NAME = 'pravin-web-app'
    }

    stages {
        stage('Build') {
            steps { 
                echo 'Build Docker image ...'
                docker.build(${env.IMAGE_NAME})
            }   
        }

        stage('Deploy to kubernetes') {
            steps {
                script {
                    sh ' kubectl apply -f kubernetes/deployment.yaml'
                    sh 'kubectl rollout status deployment/pravin-web-app'
                }
            }


        }
    }
}

  - build
  - push
  - deploy

  stage: build
  script:
    - docker build -t pathnex-web-app .

    - docker login -u "$CI_REGISTRY_USER" -p "$CI_REGISTRY_PASSWORD"

  stage: deploy

    - kubectl --context=aws-cluster apply -f kubernetes/deployment.yaml
