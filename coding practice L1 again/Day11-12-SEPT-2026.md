### Kubernetes rolling update strategy

apiVersion: app/v1
kind: Deployment
metadata:
  name: pravin-rolling
spec:
  replicas: 3
  strategy:
    type: RollingUpdate
      maxSurge: 1
      maxUnavailable: 1
  selector: 
    matchLabels:
      app: roll
  template:
    metadata:
      labels:
        app: roll
    spec:
      containers:
        - name: web
          image: nginx

    agent any
    environment {
        INSTITUTE_NAME = "Pathnex"
    }
    stages {
        stage('Checkout') {
            steps {
                checkout scm
            }
        }
        stage('Build') {
            steps {

        }
    }

## 🔹 GitLab CI — Branch Specific Jobs
You will learn how to **run jobs for specific branches in GitLab CI**.

stages:
  - build

build-main:
  stage: build
  image: maven:3.8.1-jdk-17
  script:
    - mvn clean compile
  only:
    - main

build-dev:
  stage: build
  image: maven:3.8.1-jdk-17
  script:
    - mvn clean compile
  only:
    - dev


  ###  Github Action

  name: Maven Branch Build

on:
  push:
    branches:
      - main
      - dev

jobs:
  build:
    runs-on: ubuntu-latest

    steps:
      - name: Checkout code
        uses: actions/checkout@v4

      - name: Set up JDK 17
        uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '17'
          cache: maven

      - name: Maven Build
        run: mvn clean compile
