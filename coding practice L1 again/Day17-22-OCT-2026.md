## 🔹 Ansible — Create Multiple Users in Loop

 By Pravin More


- name: Create Multiple User
  hosts: all
  become: yes

  vars:
    users:
      - dev1
      - dev2
      - dev3

  tasks:
    - name: Create USer
      user:
        name: "{{ item }}"
        state: present
      loop: "{{ users }}"
