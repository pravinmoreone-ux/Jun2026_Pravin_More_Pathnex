## 🔹 Kubernetes — Add Liveness & Readiness Probes
-- by Pravin More


apiVersion: v1
kind: Pod
metadata: 
  name: pravin-probe
spec: 
  containers:
    - name: web
      image: nginx
    
    #liveness
    livenessProbe:
      httpGet:
        path: /
        port: 80
      initialDelaySeconds: 5
      periodSeconds: 10
      failureThreshold: 3

    readinessProbe: 
      httpGet: 
        path: / 
        port: 80 
      initialDelaySeconds: 3 
      periodSeconds: 5 
      failureThreshold: 3 
      
      # Checks whether the application has finished starting. # While startupProbe is failing, liveness/readiness probes # are not executed. 
    startupProbe: 
      httpGet: 
        path: / 
        port: 80 
      periodSeconds: 5 
      failureThreshold: 12

###  jenkins

pipeline {
    agent any
    parameters {
        string(name: 'ENVIRONMENT',
               defaultValue: 'dev', 
               description: 'Deployment Environment')
    }

    ### Defines a global environment variable.
    environment {
        INSTITUTE_NAME = "Pravin"
    }

    stages {
        stage ('Checkout') { ## Downloads source code from GitHub.
            steps {
                git branch: 'main', url: 'https://github.com/Pathnex/sample-java-app.git'
            }
        }

        stage ('Build') {
            when {
                branch 'main'  # Build stage runs only if current branch is:main
            }
            steps {
                sh 'mvn clean compile' # Runs Maven compile:
            }
        }
        stage('Deploy') {
            when {
                expression { param.ENVIRONMENT == 'dev' } ##Deploy runs only when dev
            }
            steps {
                sh 'echo "Deploying to $ENVIRONMENT by $INSTITUTE_NAME"'   #vDeploying to dev by Pathnex
            }
        }
    }
}

## 🔹 GitLab CI — Conditional Jobs by pravin
You will learn how to **run jobs based on variables or branches**.

stages: 
  - build
  - deploy

build:
  stage: build
  image: maven:3.8.1-jdk-17
  script:
    - git clone https://github.com/Pathnex/sample-java-app.git
    - cd sample-java-app
    - mvn clean compile
  only:
    - main

deploy:
  stage: deploy
  image: alpine:latest
  script:
    - echo "Deploying to $CI_ENVIRONMENT_NAME by Pathnex"
  only:
    variables:
      - $CI_ENVIRONMENT_NAME == "dev"
